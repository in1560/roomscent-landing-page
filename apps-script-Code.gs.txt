var SHEET_ID = '1EcyxPNox6z11zNu-Js-uLJyfvSG2kdrvIG90k0bZ9Jo'; // เปลี่ยนตรงนี้ถ้าย้ายไปชีตใหม่

function doPost(e) {
  var sheet = SpreadsheetApp.openById(SHEET_ID).getActiveSheet();
  var data = JSON.parse(e.postData.contents);

  sheet.appendRow([
    new Date(),
    data.customerName,
    data.contact,
    data.items,
    data.total,
    data.note
  ]);

  return ContentService
    .createTextOutput("Success")
    .setMimeType(ContentService.MimeType.TEXT);
}
